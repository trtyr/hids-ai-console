## 七、文档自检脚本

```bash
node -e '
const fs = require("fs");
const text = fs.readFileSync("docs/modules/knowledge-base.md", "utf8");
const allowed = new Set(["wiki-store.js","wiki-workbench.js","wiki-replay.js"]);
const re = /([A-Za-z0-9_.\/-]+\.(?:js|yaml|json|md))[`\x27]?:(\d+)(?:-(\d+))?/g;
let m, refs = 0, bad = 0, outside = new Set();
while ((m = re.exec(text))) {
  const f = m[1], a = +m[2], b = m[3] ? +m[3] : a;
  refs++;
  if (!fs.existsSync(f)) { console.log("缺文件:", f); bad++; continue; }
  if (!allowed.has(f)) { outside.add(f); continue; }
  const fl = fs.readFileSync(f, "utf8").split("\n").length;
  if (a < 1 || b > fl || a > b) { console.log("行号越界:", f+":"+a+"-"+b, "(共"+fl+"行)"); bad++; }
}
console.log("path:line 引用:", refs, "| 越界或缺文件:", bad, "| 清单外文件:", outside.size ? [...outside].join(", ") : "无");
const lines = text.split("\n");
let tbad = 0, inTable = false, cols = 0;
for (const l of lines) {
  if (/^\|/.test(l)) { const c = l.split("|").length;
    if (!inTable) { inTable = true; cols = c; } else if (c !== cols) { console.log("列数不一致:", l.slice(0,60), c, "vs", cols); tbad++; }
  } else inTable = false;
}
console.log("表格列数问题:", tbad);
'
```
