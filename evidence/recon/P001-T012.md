## 十、引用核对脚本

放在本文件里，方便下一棒直接复制去跑：

```bash
node -e '
const fs=require("fs");
const page="docs/modules/prompt-store.md";
const manifest="evidence/recon/P001-T012.md";
const doc=fs.readFileSync(page,"utf8");
const man=fs.readFileSync(manifest,"utf8");
const scope=["prompt-store.js"];                 // _breakdown.json 的 prompt-store.files
const src=fs.readFileSync("prompt-store.js","utf8").split("\n");
let m,refs=0,bad=0; const used=new Map();
const re=/([A-Za-z0-9_.\/-]+\.(?:js|yaml|json|md))[`\x27]?:(\d+)(?:-(\d+))?/g;
while((m=re.exec(doc))){
  const f=m[1],a=+m[2],c=m[3]?+m[3]:a; refs++; used.set(f+":"+a+(m[3]?"-"+c:""),(used.get(f+":"+a+(m[3]?"-"+c:""))||0)+1);
  if(!scope.includes(f)){console.log("清单外引用:",f+":"+a);bad++;continue;}
  if(a<1||c>src.length){console.log("行号越界:",f+":"+a+"-"+c);bad++;}
}
const table=man.split("## 九、锚点清单")[1].split("## 十、")[0];   // 只认清单表里的锚点
const listed=new Set([...table.matchAll(/`(prompt-store\.js:\d+(?:-\d+)?)`/g)].map(x=>x[1]));
let miss=0,unused=0;
for (const k of used.keys()) if(!listed.has(k)){console.log("页面引用未登记:",k);miss++;}
for (const k of listed) if(!used.has(k)){console.log("登记了但页面没用:",k);unused++;}
console.log("页面引用处数:",refs,"| 不同锚点:",used.size,"| 清单条目:",listed.size,
            "| 清单外:",bad,"| 未登记:",miss,"| 未用:",unused);
process.exit(bad||miss||unused?1:0);
'
```
