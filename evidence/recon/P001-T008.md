import { readFileSync } from 'node:fs';
const doc = readFileSync('docs/modules/triage-pipeline.md', 'utf8');

### 5.1 引用机器核对（复核后重跑）

```bash
cat > /tmp/week-refcheck.mjs <<'EOF'
import { readFileSync } from 'node:fs';
const page = readFileSync('docs/modules/triage-pipeline.md','utf8');
const files = ['triage-monitor.js','syslog-input.js','syslog-case.js'];
const len = Object.fromEntries(files.map(f=>[f, readFileSync(f,'utf8').split('\n').length]));
const re = /([A-Za-z0-9_.\/-]+\.(?:js|json|yaml|md|txt)):(\d+)(?:-(\d+))?/g;
let total=0, outOfScope=[], outOfBounds=[], perFile={};
for (const m of page.matchAll(re)) {
  total++;
  const [_, file, a, b] = m;
  const base = file.split('/').pop();
  perFile[file]=(perFile[file]||0)+1;
  if (!files.includes(base)) { outOfScope.push(`${file}:${a}`); continue; }
  const n = Math.max(Number(a), Number(b||a));
  if (n > len[base]) outOfBounds.push(`${file}:${n} > ${len[base]}`);
}
console.log('引用 token 总数(全量):', total);
console.log('按文件:', JSON.stringify(perFile));
console.log('清单外文件引用:', outOfScope.length, outOfScope.slice(0,10));
console.log('越界行号:', outOfBounds.length, outOfBounds.slice(0,10));
console.log('行数:', JSON.stringify(len));
EOF
node /tmp/week-refcheck.mjs
```
