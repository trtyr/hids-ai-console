## 四、核对方法

页面里的引用必须与本清单逐字对应（同一行号区间写法）。核对脚本：

```bash
node -e '
const fs = require("fs");
const page = fs.readFileSync("docs/modules/trace-board.md", "utf8");
const manifest = fs.readFileSync("evidence/recon/P001-T011.md", "utf8");
const anchors = new Set([...manifest.matchAll(/\|\s*`([A-Za-z0-9_.\/-]+\.[A-Za-z0-9]+:[0-9]+(?:-[0-9]+)?)`/g)].map(m => m[1]));
const refs = [...page.matchAll(/([A-Za-z0-9_.\/-]+\.[A-Za-z0-9]+):([0-9]+)(?:-([0-9]+))?/g)];
let missing = 0;
for (const m of refs) {
  const key = `${m[1]}:${m[2]}` + (m[3] ? `-${m[3]}` : "");
  if (!anchors.has(key)) { missing++; console.log("清单外引用:", key); continue; }
  if (!fs.existsSync(m[1])) { missing++; console.log("文件不存在:", m[1]); continue; }
  const lines = fs.readFileSync(m[1], "utf8").split("\n").length;
  const end = m[3] ? Number(m[3]) : Number(m[2]);
  if (Number(m[2]) < 1 || end > lines) { missing++; console.log("行号越界:", key, "文件行数", lines); }
}
console.log("页面引用处数:", refs.length, "| 清单条目:", anchors.size, "| 问题:", missing);
process.exit(missing ? 1 : 0);
'
```

