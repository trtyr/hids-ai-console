```bash
cat > /tmp/check-page.mjs <<'EOF'
import { readFileSync, existsSync } from 'node:fs';
const page = readFileSync('docs/modules/record-store.md', 'utf8');
const man = readFileSync('evidence/recon/P001-T014.md', 'utf8');
const refs = new Set(), bad = [];
for (const m of page.matchAll(/([A-Za-z0-9_./-]+\.(?:js|json|md)):(\d+)(?:-(\d+))?/g)) {
  const key = `${m[1]}:${m[2]}${m[3] ? '-' + m[3] : ''}`;
  if (refs.has(key)) continue;
  refs.add(key);
  if (!existsSync(m[1])) { bad.push(`${key} 文件不存在`); continue; }
  const n = readFileSync(m[1], 'utf8').split('\n').length;
  const a = +m[2], b = m[3] ? +m[3] : a;
  if (a < 1 || b > n) bad.push(`${key} 行号越界(共${n}行)`);
}
const inMan = new Set([...man.matchAll(/`([A-Za-z0-9_./-]+\.(?:js|json|md):\d+(?:-\d+)?)`/g)].map(m => m[1]));
const outside = [...refs].filter(r => !inMan.has(r));
console.log('引用总数:', refs.size);
console.log('路径/行号不成立:', bad.length ? bad.join('; ') : 0);
console.log('不在清单覆盖表内:', outside.length ? outside.join(', ') : 0);
process.exit(bad.length || outside.length ? 1 : 0);
EOF
node /tmp/check-page.mjs; echo "exit=$?"
```

```

不在清单覆盖表内: 0
exit=0
```
