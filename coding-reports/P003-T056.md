## 3. 怎么验的（可复跑脚本）

```python
# verify_refs.py —— 从报告 §4 的表里读「旧/新/符号」，逐条真跑 grep -nF，断言命中行落在新区间内
import re, subprocess, sys
report, src = sys.argv[1], sys.argv[2]
rows = []
for line in open(report, encoding='utf8'):
    m = re.match(r'\| \d+ \| `:([\d-]+)` \| `:([\d-]+)`[^|]*\| `(.+?)` \|', line)
    if m: rows.append(m.groups())
def span(s):
    p = [int(x) for x in s.split('-')]; return p[0], p[-1]
bad = []
for old, new, sym in rows:
    ns, ne = span(new)
    out = subprocess.run(['grep', '-nF', '--', sym, src], capture_output=True, text=True).stdout
    hits = [int(l.split(':', 1)[0]) for l in out.splitlines() if l]
    if not any(ns <= h <= ne for h in hits): bad.append((old, new, sym, hits))
print(f'rows {len(rows)} fails {len(bad)}')
for b in bad: print('FAIL', b)
sys.exit(1 if bad or not rows else 0)
```
